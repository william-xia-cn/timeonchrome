```javascript
  await cleanExpiredTempWhitelist();

  // studyList + allowList + 临时白名单 都放行
  const studyList  = cfg.studyList  || [];
  const allowList  = cfg.allowList  || [];

  const allAllowed = [...studyList, ...allowList, ...tempDomains];

  if (removeIds.length > 0) {
    await chrome.declarativeNetRequest.updateDynamicRules({ removeRuleIds: removeIds });
  }

  if (allAllowed.length === 0) return;

  const rules = [];
  let ruleId = 1000;
  for (const domain of allAllowed) {

      id: ruleId++,
      priority: 1,
      action: { type: 'allow' },
      condition: { urlFilter: `||${domain}^`, resourceTypes: ['main_frame'] }
    });
  }
  await chrome.declarativeNetRequest.updateDynamicRules({ addRules: rules });
  return;
}
```

**Files:**

- [ ] **Step 1: HEARTBEAT 按域名类型分别计时**

用以下代码替换现有 `case 'HEARTBEAT'` 块：

```javascript
case 'HEARTBEAT': {
  if (!sender.tab || !sender.tab.active) return { ok: true };
  const domain = extractDomain(sender.tab.url);
  if (!domain) return { ok: true };

  const TICK = 10;
  const cfg = await getConfig();
  const session = await getSession();

  const inStudyList = (cfg.studyList || []).some(w => matchDomain(domain, w));
  const inAllowList = (cfg.allowList || []).some(w => matchDomain(domain, w));

  // 1. 域名时长：active 和 passive 都计
  if (msg.state === 'active' || msg.state === 'passive') {
    await addDomainTime(domain, TICK);
  }

  // 2. 在线时长 / 学习时长（仅 active，排除 passive 音乐）
  if (msg.state === 'active') {
    if (session.currentMode === 'study' && inStudyList) {
      // 学习网站 active → 学习时长
      session.studySession.totalSeconds += TICK;
      await saveSession(session);
    } else if (session.currentMode === 'study' && inAllowList) {
      // 允许网站 active → 只计域名时长，不计学习（已在步骤1计入）
    } else if (session.currentMode === 'rest') {
      // 娱乐模式 active → 娱乐时长
      session.restSession.totalSeconds += TICK;
      await saveSession(session);
    }

    // 3. 自动切换学习模式
    handleAutoStudyTick(domain, cfg, session);
  }

  return { ok: true };
}
```

- [ ] **Step 2: 新增 handleAutoStudyTick 函数**

在 `blockTab` 函数附近新增：

```javascript
function handleAutoStudyTick(domain, config, session) {
  // 只在 rest 模式下检测
  if (session.currentMode !== 'rest') {
    autoStudyCounter = 0;
    return;
  }

  if (!config.autoStudyConfig?.enabled) return;

  const inStudyList = (config.studyList || []).some(w => matchDomain(domain, w));
  const inAllowList = (config.allowList || []).some(w => matchDomain(domain, w));
  const now = Date.now();
  const TICK = 10;
  const required = config.autoStudyConfig.requiredSeconds || 90;
  const MAX_GAP_MS = 120 * 1000; // 超过 2 分钟无 tick，重置

  if (inStudyList) {
    // 连续性检查：上次 tick 超过 2 分钟则重置
    if (autoStudyLastTick > 0 && now - autoStudyLastTick > MAX_GAP_MS) {
      autoStudyCounter = 0;
    }
    autoStudyCounter += TICK;
    autoStudyLastTick = now;

    if (autoStudyCounter >= required) {
      autoStudyCounter = 0;
      autoStudyLastTick = 0;
      // 异步切换，不 await（避免阻塞 HEARTBEAT 返回）
      switchToStudy().then(() => {
        chrome.notifications.create({
          type: 'basic',
          iconUrl: 'icons/icon48.png',
          title: '家长守护',
          message: '检测到你在学习，已自动切换到学习模式 📚'
        });
      }).catch(() => {});
    }
  } else if (inAllowList) {
    // 允许网站：暂停计数（更新时间戳但不累加，也不重置）
    autoStudyLastTick = now;
  } else {
    // 非白名单网站：重置
    autoStudyCounter = 0;
    autoStudyLastTick = 0;
  }
}
```

```javascript
async function updateTodayStats(sessions) {
  const today = new Date().toISOString().split('T')[0];
  const todayData = sessions[today] || { studySeconds: 0, restSeconds: 0 };

  if (currentSession) {
    todayData.studySeconds += currentSession.studySession.totalSeconds;
    todayData.restSeconds  += currentSession.restSession.totalSeconds;
  }

  const onlineSeconds = todayData.studySeconds + todayData.restSeconds;

  document.getElementById('study-time-stat').textContent  = formatSeconds(todayData.studySeconds);
  document.getElementById('rest-time-stat').textContent   = formatSeconds(todayData.restSeconds);
  document.getElementById('online-time-stat').textContent = formatSeconds(onlineSeconds);
}
```

```javascript
function loadAutoStudyConfig() {
  const cfg = config.autoStudyConfig || { enabled: true, requiredSeconds: 90 };
  document.getElementById('auto-study-enabled').checked  = cfg.enabled;
  document.getElementById('auto-study-seconds').value    = cfg.requiredSeconds;
}

function setupAutoStudyConfigSave() {
  const btn = document.getElementById('save-auto-study-btn');
  if (btn.dataset.bound) return;
  btn.dataset.bound = true;

  btn.addEventListener('click', async () => {
    config.autoStudyConfig = {
      enabled:         document.getElementById('auto-study-enabled').checked,
      requiredSeconds: parseInt(document.getElementById('auto-study-seconds').value) || 90
    };
    await sendMsg({ type: 'UPDATE_CONFIG', config });
    showSuccess();
  });
}
```
